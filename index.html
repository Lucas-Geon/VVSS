// --- 1. 알람 사운드 로드 (동일 폴더 내 alarm.mp3) --- const alarmAudio = new Audio('./alarm.mp3'); alarmAudio.loop = true; // 알람 해제 전까지 반복 재생 let isAlarmActive = false; // --- 2. 심전도 파라미터 상태 객체 예시 --- const ecgState = { stDeviation: 0.0, // 단위: mV (양수: Elevation, 음수: Depression) tWaveInverted: false, // T-wave Inversion 플래그 (또는 T wave amplitude < 0) tWaveAmplitude: 0.2 // 단위: mV }; // --- 3. 알람 상태 평가 함수 --- function evaluateEcgAlarms() { const activeAlerts = []; // 1) ST Depression: -1.0 mV 이하 if (ecgState.stDeviation <= -1.0) { activeAlerts.push({ type: 'ST_DEPRESSION', message: `ST Depression Warning (${ecgState.stDeviation.toFixed(1)} mV)` }); } // 2) ST Elevation: +2.0 mV 이상 if (ecgState.stDeviation >= 2.0) { activeAlerts.push({ type: 'ST_ELEVATION', message: `ST Elevation Critical (${ecgState.stDeviation.toFixed(1)} mV)` }); } // 3) T-wave Inversion: 플래그 활성화 또는 진폭 음수 if (ecgState.tWaveInverted || ecgState.tWaveAmplitude < 0) { activeAlerts.push({ type: 'T_INVERSION', message: 'T-Wave Inversion Detected' }); } // 알람 UI 및 오디오 제어 updateAlarmStatus(activeAlerts); } // --- 4. 알람 UI 및 오디오 처리 --- function updateAlarmStatus(alerts) { const alarmDisplay = document.getElementById('alarm-display'); // 알람 메시지 컨테이너 if (alerts.length > 0) { // 알람 발생 상태 if (!isAlarmActive) { isAlarmActive = true; alarmAudio.play().catch(e => console.log('Audio autoplay prevented by browser policy:', e)); } // UI에 경고 표시 if (alarmDisplay) { alarmDisplay.innerHTML = alerts.map(a => `
${a.message}
`).join(''); alarmDisplay.classList.add('active'); } } else { // 정상 상태 복귀 시 알람 정지 if (isAlarmActive) { isAlarmActive = false; alarmAudio.pause(); alarmAudio.currentTime = 0; // 재생 위치 초기화 } if (alarmDisplay) { alarmDisplay.innerHTML = ''; alarmDisplay.classList.remove('active'); } } } // --- 5. 수치 변경/인풋 이벤트 연동 예시 --- function setEcgParameters(stVal, tAmpVal, tInvertFlag) { ecgState.stDeviation = parseFloat(stVal); ecgState.tWaveAmplitude = parseFloat(tAmpVal); ecgState.tWaveInverted = Boolean(tInvertFlag); evaluateEcgAlarms(); }
